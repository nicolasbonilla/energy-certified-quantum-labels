% fig_map_rd.tex -- main Fig. 3 on the PUBLIC data (2026-11-12): the certification map at the published energies.
% Data: dat/*.dat, written by export_map_rd.py from phase3/full_runs/real_data (out/map_rows.json,
% e1/map_e1/map_rows.json) and the stored emulation outputs. Compile in paper/figures/pgf_map_rd/ with pdflatex
% (build_rd.py). Earlier version: ../_rd_v1_before_F2/pgf_map_rd/fig_map_rd.tex.
% 2026-11-13 (F2, editor-C-12/13): one status code shared with Figs. 1(c) and 2(b) -- certified = the teal of Fig. 1
% (dark rim), not certified = dark grey, undecided = light grey with a grey rim (hatched cells), certified only with
% the extension E1 = light teal with a dot; colour marks the status only, the label is marked by symbol shape and
% line style; the third average and the four local labels are drawn in greys; strip rows named in words at the size of
% the tick labels; undecided bars drawn up to at most 3 tau (an arrow marks a longer bar, dat/ucap_*.dat).

% #1 = n, #2 = file key, #3 = line style, #4 = marker style. The line joins neighbouring not-certified couplings
% (column N); the markers are those of column Nm (every coupling for m_s and Czz_bar; for the other labels every
% third coupling, staggered between labels, and every not-certified point without a not-certified neighbour)
\newcommand{\notcert}[4]{%
  \addplot[#3, forget plot] table[x=J, y=N] {dat/w_n#1_#2.dat};
  \addplot[#4, forget plot] table[x=J, y=Nm] {dat/w_n#1_#2.dat};}
\newcommand{\cert}[4]{% #1 = n, #2 = key, #3 = line, #4 = marker
  \addplot[#3, forget plot] table[x=J, y=C] {dat/w_n#1_#2.dat};
  \addplot[#4, forget plot] table[x=J, y=C] {dat/w_n#1_#2.dat};}
\newcommand{\undec}[3]{% #1 = n, #2 = key, #3 = mark shape; bars end at 3 tau at most (arrow: dat/ucap_*.dat)
  \addplot[cUndRim, line width=0.6pt, forget plot] table[x=J, y=w] {dat/ubar_n#1_#2.dat};
  \addplot[only marks, mark=triangle*, mark size=1.7pt, mark options={solid, draw=cUndRim, fill=cUndRim}, forget plot]
    table[x=J, y=w] {dat/ucap_n#1_#2.dat};
  \addplot[only marks, mark=#3, mark size=1.3pt, mark options={solid, draw=cUndRim, fill=cUnd, line width=0.5pt}, forget plot]
    table[x=J, y=Ul] {dat/w_n#1_#2.dat};
  \addplot[only marks, mark=-, mark size=1.6pt, mark options={solid, draw=cUndRim, line width=0.6pt}, forget plot]
    table[x=J, y=Uud] {dat/w_n#1_#2.dat};}
\newcommand{\extE}[3]{% #1 = n, #2 = key, #3 = mark shape: the layer of the extension E1
  \addplot[cCert, line width=0.5pt, densely dashed, forget plot] table[x=J, y=E1] {dat/w_n#1_#2.dat};
  \addplot[only marks, mark=#3, mark size=1.5pt, mark options={solid, draw=cCert, fill=cCertE, line width=0.55pt}, forget plot]
    table[x=J, y=E1] {dat/w_n#1_#2.dat};}

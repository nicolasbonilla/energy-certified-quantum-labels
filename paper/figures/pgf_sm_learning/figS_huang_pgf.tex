% figS_huang_pgf.tex -- Supplemental figure of the second benchmark (fig:huang) as a native PGFPlots figure.

% Panel (d) (pair classes, all single pairs) uses a separate categorical set that avoids teal, orange and grey.
\documentclass[10pt,border=1pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{amsmath}

\pgfplotsset{compat=1.18}
\usetikzlibrary{calc}
\definecolor{cnn}{HTML}{169B88}    % average, nearest neighbours (teal)
\definecolor{cd2}{HTML}{0B5F53}    % average, diagonal pairs (dark teal)
\definecolor{ccent}{HTML}{CF5521}  % central bond (orange)
\definecolor{cfar}{HTML}{8A3A12}   % end-to-end pair (dark orange-brown)
\definecolor{cfix}{HTML}{E08E0B}   % fixed bond (7,12) (amber)

\pgfplotsset{
  every axis/.append style={
    scale only axis, tick label style={font=\footnotesize}, label style={font=\small},
    axis line style={line width=0.5pt}, every tick/.style={line width=0.5pt, black}, tick align=inside,
    tick pos=left, legend cell align=left, unbounded coords=jump,
    legend style={font=\footnotesize, draw=none, fill=none, inner sep=1.5pt, row sep=0.5pt},
  },
  every axis plot/.append style={line join=round},
  wax/.style={width=7.3cm, height=4.6cm, xmin=8.5e-4, xmax=1.18e-2, ymin=0.07, ymax=8,
    xtick={0.001,0.002,0.005,0.01}, xticklabels={0.001,0.002,0.005,0.01},
    ytick={0.1,0.2,0.5,1,2,5}, yticklabels={0.1,0.2,0.5,1,2,5}, xminorticks=false, yminorticks=false,
    xlabel={energy error per spin $\delta e$}, ylabel={width$\,/\,\tau$}},
  Cnn/.style={cnn, mark=*, mark size=1.7pt},
  Cd2/.style={cd2, mark=square*, mark size=1.55pt},
  Ccent/.style={ccent, mark=triangle*, mark size=2.1pt},
  Cfar/.style={cfar, mark=diamond*, mark size=2.1pt},
  Cfix/.style={cfix, mark=triangle*, mark size=2.1pt, mark options={rotate=180}},
  outer/.style={line width=1.0pt},
  inner/.style={line width=0.8pt, dash pattern=on 3pt off 2pt, no markers},
  markimg/.style={legend image code/.code={\path[draw=none, fill=none] (0cm,0cm) -- (0.6cm,0cm);
    \draw[##1] plot coordinates {(0.3cm,0cm)};}},
}
\newcommand{\wshade}{\fill[band] (axis cs:1e-3,0.07) rectangle (axis cs:1e-2,8);}
\newcommand{\wtau}{\draw[ink, line width=0.8pt] (axis cs:8.5e-4,1) -- (axis cs:1.18e-2,1);}
% (c): one label; #1 = label key
\newcommand{\cset}[1]{%
  \addplot[#1, no markers, line width=0.9pt, forget plot] table[x=x,y=de] {hu_c_#1_bars.dat};
  \addplot[#1, only marks, forget plot, mark options={fill=white, line width=0.6pt}, mark size=2.9pt]
    table[x=x,y=de_law] {hu_c_#1.dat};
  \addplot[#1, only marks, forget plot] table[x=x,y=de_out] {hu_c_#1.dat};
}
